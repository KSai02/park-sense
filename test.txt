  metro-app:
    image: <your-dockerhub-username>/metro-booking-app:v1
    container_name: metro-app
    ports:
      - "8080:8080"           # Host:Container port mapping
    depends_on:
      - db
    environment:
      # Example environment variable (if needed)
      DB_HOST: db
      DB_PORT: 3306
      DB_USER: root
      DB_PASSWORD: root
    restart: always

  db:
    image: mysql:8.0
    container_name: metro-db
    environment:
      MYSQL_ROOT_PASSWORD: root
      MYSQL_DATABASE: metro
