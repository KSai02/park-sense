# Use a base JDK image
FROM openjdk:17-jdk-slim

# Set working directory
WORKDIR /app

# Copy the built jar file into the container
COPY target/metro-booking-app.jar metro-booking-app.jar

# Expose application port
EXPOSE 8080

# Run the application
ENTRYPOINT ["java","-jar","metro-booking-app.jar"]


////////////////////////////////////////////////////////////////////////////////
First, build your Maven project:

mvn clean package


Then build the Docker image:

docker build -t metro-booking-app .

3️⃣ Push to Docker Hub (5 Marks)

Login to Docker Hub

docker login


Tag the image

docker tag metro-booking-app <your-dockerhub-username>/metro-booking-app:v1


Push the image

docker push <your-dockerhub-username>/metro-booking-app:v1
